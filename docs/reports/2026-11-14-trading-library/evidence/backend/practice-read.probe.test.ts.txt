type Metric = { calls:number; rows:number; textBytes:number; blobBytes:number; sql:Record<string,{calls:number;rows:number;textBytes:number;blobBytes:number}> };
const empty = ():Metric => ({calls:0,rows:0,textBytes:0,blobBytes:0,sql:{}});
function meter(db:DatabaseSync) {
  let current = empty();
  const proxy = new Proxy(db, { get(target,key) {
    if(key === 'prepare') return (sql:string) => {
      const statement = target.prepare(sql);
      return new Proxy(statement, {get(s,k) {
        if(k === 'all' || k === 'get') return (...args:unknown[]) => {
          const result = (s[k] as Function).apply(s,args);
          const rows = k==='all' ? result : result ? [result] : [];
          const local = current.sql[sql] ??= {calls:0,rows:0,textBytes:0,blobBytes:0};
          current.calls++; local.calls++; current.rows += rows.length; local.rows += rows.length;
          for(const r of rows) for(const v of Object.values(r)) {
            if(typeof v==='string') { const n=Buffer.byteLength(v); current.textBytes+=n;local.textBytes+=n; }
            if(v instanceof Uint8Array) {current.blobBytes+=v.byteLength;local.blobBytes+=v.byteLength;}
          }
          return result;
        };
        const value=Reflect.get(s,k,s);return typeof value==='function'?value.bind(s):value;
      }});
    };
    const value=Reflect.get(target,key,target);return typeof value==='function'?value.bind(target):value;
  }});
  return {proxy,reset(){current=empty();},result(){return structuredClone(current);}};
}
