test('replay actual readiness and page against existing isolated A1b publication',async()=>{
  const path=realpathSync('/Users/zhoulin/.codex/worktrees/38a0/TradeReview/.scratch/homepage-db-optimization-20261105/module-implementation-20261106/acceptance-trades-a1b.sqlite');

  async function sequence(){
    const at=performance.now();let t=at;
    const ready=await POST(request({operation:'readiness'}));const readiness=JSON.parse(await ready.text());
    const readinessMs=performance.now()-t;t=performance.now();expect(ready.status).toBe(200);

  }
  try {
    const head=db.prepare("select publication_id from homepage_read_head where product='trades'").get()!;
    const persisted=Object.fromEntries([['details','trade_library_round_details','payload_json'],['evidence','trade_library_evidence_refs','target_json']].map(([name,table,column])=>[name,db.prepare(`select count(*) rows,sum(length(cast(${column} as blob))) bytes from ${table} where publication_id=?`).get(head.publication_id)]));

  } finally {vi.useRealTimers();connection.database=undefined;db.close();}
